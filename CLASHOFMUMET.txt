        /* ==========================================================================
           BAGIAN 2: STATE MANAGEMENT & VARIABLES
           ========================================================================== */
        let gameState = {
            difficulty: "mudah",
            mode: "balapan",
            speed: 20, // detik
            currentQuestions: [],
            currentIndex: 0,
            score: 0,
            timerInterval: null,
            timeLeft: 0,
            recap: [],
            isAnsweringBlocked: false
        };

        /* ==========================================================================
           BAGIAN 3: AUDIO SYNTHESIZER (WEB AUDIO API)
           ========================================================================== */
        let audioCtx = null;

        function initAudio() {
            if (!audioCtx) {
                audioCtx = new (window.AudioContext || window.webkitAudioContext)();
            }
        }

        function playSound(type) {
            initAudio();
            if (!audioCtx) return;

            const osc = audioCtx.createOscillator();
            const gain = audioCtx.createGain();
            osc.connect(gain);
            gain.connect(audioCtx.destination);

            if (type === 'correct') {
                // Beep Frekuensi Tinggi (Sukses)
                osc.type = 'sine';
                osc.frequency.setValueAtTime(600, audioCtx.currentTime);
                osc.frequency.exponentialRampToValueAtTime(1200, audioCtx.currentTime + 0.15);
                gain.gain.setValueAtTime(0.2, audioCtx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.15);
                osc.start(audioCtx.currentTime);
                osc.stop(audioCtx.currentTime + 0.15);
            } else if (type === 'wrong') {
                // Buzz Frekuensi Rendah (Gagal)
                osc.type = 'sawtooth';
                osc.frequency.setValueAtTime(180, audioCtx.currentTime);
                osc.frequency.exponentialRampToValueAtTime(90, audioCtx.currentTime + 0.25);
                gain.gain.setValueAtTime(0.25, audioCtx.currentTime);
                gain.gain.exponentialRampToValueAtTime(0.01, audioCtx.currentTime + 0.25);
                osc.start(audioCtx.currentTime);
                osc.stop(audioCtx.currentTime + 0.25);
            }
        }

        /* ==========================================================================
           BAGIAN 4: NAVIGASI STATE / HALAMAN
           ========================================================================== */
        function navigateTo(pageId) {
            document.querySelectorAll('.page').forEach(page => {
                page.classList.remove('active');
            });
            document.getElementById(pageId).classList.add('active');
        }

        /* ==========================================================================
           BAGIAN 5: KONFIGURASI SELECTION LOGIC
           ========================================================================== */
        function selectChip(element) {
            const group = element.getAttribute('data-group');
            document.querySelectorAll(`.chip-option[data-group="${group}"]`).forEach(chip => {
                chip.classList.remove('selected');
            });
            element.classList.add('selected');

            // Save to state
            const val = element.getAttribute('data-value');
            if (group === 'difficulty') gameState.difficulty = val;
            if (group === 'speed') gameState.speed = parseInt(val);
        }

        function selectMode(element, modeValue) {
            selectChip(element);
            gameState.mode = modeValue;

            const speedSection = document.getElementById('speed-section');
            if (modeValue === 'balapan') {
                speedSection.style.display = 'block';
            } else {
                speedSection.style.display = 'none';
            }
        }

        /* ==========================================================================
           BAGIAN 6: UTILITY (SHUFFLE)
           ========================================================================== */
        function shuffleArray(array) {
            let arr = [...array];
            for (let i = arr.length - 1; i > 0; i--) {
                const j = Math.floor(Math.random() * (i + 1));
                [arr[i], arr[j]] = [arr[j], arr[i]];
            }
            return arr;
        }

        /* ==========================================================================
           BAGIAN 7: ARENA GAMEPLAY LOGIC
           ========================================================================== */
        function initQuizArena() {
            // Get database based on difficulty
            const rawQuestions = quizDatabase[gameState.difficulty];
            gameState.currentQuestions = shuffleArray(rawQuestions);
            gameState.currentIndex = 0;
            gameState.score = 0;
            gameState.recap = [];
            gameState.isAnsweringBlocked = false;

            // UI Adjustments
            const timerContainer = document.getElementById('timer-container');
            const timerHud = document.getElementById('hud-timer-box');
            
            if (gameState.mode === 'balapan') {
                timerContainer.style.display = 'block';
                timerHud.style.display = 'flex';
            } else {
                timerContainer.style.display = 'none';
                timerHud.style.display = 'none';
            }

            navigateTo('page-arena');
            loadQuestion();
        }

        function loadQuestion() {
            clearInterval(gameState.timerInterval);
            gameState.isAnsweringBlocked = false;

            const q = gameState.currentQuestions[gameState.currentIndex];
            
            // Update HUD
            document.getElementById('hud-question-number').innerText = `${gameState.currentIndex + 1}/20`;
            document.getElementById('hud-score').innerText = gameState.score;
            document.getElementById('question-text').innerText = q.teks;

            // Render Answers
            const grid = document.getElementById('answers-grid');
            grid.innerHTML = '';
            const prefixes = ['A', 'B', 'C', 'D'];

            q.pilihan.forEach((opt, idx) => {
                const prefix = prefixes[idx];
                const btn = document.createElement('button');
                btn.className = 'answer-btn';
                btn.setAttribute('data-choice', prefix);
                btn.innerHTML = `<span class="choice-prefix">${prefix}.</span> ${opt}`;
                btn.onclick = () => handleAnswer(prefix, btn);
                grid.appendChild(btn);
            });

            // Start Timer if Balapan
            if (gameState.mode === 'balapan') {
                startTimer();
            }
        }

        function startTimer() {
            gameState.timeLeft = gameState.speed;
            const timerBar = document.getElementById('timer-bar');
            const timerHud = document.getElementById('hud-timer');
            
            timerBar.style.width = '100%';
            timerHud.innerText = `${gameState.timeLeft}s`;

            const totalTime = gameState.speed;
            const intervalTime = 100; // ms
            let elapsed = 0;

            gameState.timerInterval = setInterval(() => {
                elapsed += 0.1;
                gameState.timeLeft = Math.max(0, totalTime - elapsed);
                
                // Update UI
                const percent = (gameState.timeLeft / totalTime) * 100;
                timerBar.style.width = `${percent}%`;
                timerHud.innerText = `${Math.ceil(gameState.timeLeft)}s`;

                if (gameState.timeLeft <= 0) {
                    clearInterval(gameState.timerInterval);
                    handleTimeOut();
                }
            }, intervalTime);
        }

        function handleAnswer(selectedChoice, clickedBtn) {
            if (gameState.isAnsweringBlocked) return;
            gameState.isAnsweringBlocked = true;
            clearInterval(gameState.timerInterval);

            const q = gameState.currentQuestions[gameState.currentIndex];
            const isCorrect = (selectedChoice === q.jawabanBenar);
            const allBtns = document.querySelectorAll('.answer-btn');

            allBtns.forEach(btn => btn.classList.add('disabled'));

            if (isCorrect) {
                clickedBtn.classList.add('btn-correct');
                playSound('correct');
                gameState.score += 5;
                document.getElementById('hud-score').innerText = gameState.score;
            } else {
                clickedBtn.classList.add('btn-wrong');
                playSound('wrong');
                // Highlight correct answer
                allBtns.forEach(btn => {
                    if (btn.getAttribute('data-choice') === q.jawabanBenar) {
                        btn.classList.add('btn-correct');
                    }
                });
            }

            // Save Recap
            gameState.recap.push({
                no: gameState.currentIndex + 1,
                soal: q.teks,
                isCorrect: isCorrect
            });

            // Delay 1.5s then next
            setTimeout(nextQuestion, 1500);
        }

        function handleTimeOut() {
            if (gameState.isAnsweringBlocked) return;
            gameState.isAnsweringBlocked = true;

            playSound('wrong');
            const q = gameState.currentQuestions[gameState.currentIndex];
            const allBtns = document.querySelectorAll('.answer-btn');

            allBtns.forEach(btn => {
                btn.classList.add('disabled');
                if (btn.getAttribute('data-choice') === q.jawabanBenar) {
                    btn.classList.add('btn-correct');
                }
            });

            // Save Recap as Wrong
            gameState.recap.push({
                no: gameState.currentIndex + 1,
                soal: q.teks,
                isCorrect: false
            });

            setTimeout(nextQuestion, 1500);
        }

        function nextQuestion() {
            gameState.currentIndex++;
            if (gameState.currentIndex < 20) {
                loadQuestion();
            } else {
                showResults();
            }
        }

        /* ==========================================================================
           BAGIAN 8: EVALUASI DAN HASIL AKHIR
           ========================================================================== */
        function showResults() {
            navigateTo('page-result');

            const score = gameState.score;
            document.getElementById('final-score-text').innerText = score;

            const evalTitle = document.getElementById('eval-title');
            const evalDesc = document.getElementById('eval-desc');

            if (score === 100) {
                evalTitle.innerText = "Sempurna! Biologist Sejati!";
                evalDesc.innerText = "Luar biasa! Performa kamu sempurna tanpa cela. Pemahaman taksonomi kamu sudah berada di tingkat Master Akademik sekelas Fikri Tampan!";
            } else if (score >= 80) {
                evalTitle.innerText = "Performa Unggul!";
                evalDesc.innerText = "Sangat bagus! Kamu menguasai sebagian besar konsep klasifikasi makhluk hidup dengan analisis yang tajam dan cepat.";
            } else if (score >= 60) {
                evalTitle.innerText = "Cukup Baik!";
                evalDesc.innerText = "Hasil yang lumayan, namun masih ada beberapa konsep dan aturan taksonomi yang perlu dipoles kembali agar lebih presisi.";
            } else {
                evalTitle.innerText = "Perlu Latihan Lagi!";
                evalDesc.innerText = "Jangan berkecil hati. Pelajari kembali materi dasar klasifikasi, tata nama ilmiah, dan aturan taksonomi lalu coba arena kuis ini kembali!";
            }

            // Render Recap Table
            const tbody = document.getElementById('recap-tbody');
            tbody.innerHTML = '';

            gameState.recap.forEach(item => {
                const tr = document.createElement('tr');
                const badge = item.isCorrect 
                    ? `<span class="badge-status badge-correct">Benar (+5)</span>`
                    : `<span class="badge-status badge-wrong">Salah (0)</span>`;

                tr.innerHTML = `
                    <td style="font-weight: 700;">${item.no}</td>
                    <td>${item.soal}</td>
                    <td>${badge}</td>
                `;
                tbody.appendChild(tr);
            });
        }
